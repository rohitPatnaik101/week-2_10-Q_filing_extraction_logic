import pdfplumber
import re

def extract_company_name(text):
    match = re.search(r"(.*?)\n(?=\(?Exact name of registrant as specified in its charter\)?)", text, re.IGNORECASE)
    if match:
        return match.group(1).strip()
    return None

def extract_filing_date(text):
    match = re.search(r"For the quarterly period ended\s+([A-Za-z]+\s+\d{1,2},\s+\d{4})", text)
    if match:
        return match.group(1).strip()
    return None

def find_item_2_page(pdf):
    """Extract Table of Contents and locate the page number for Item 2."""
    for page in pdf.pages[:5]:  # Usually, TOC is in the first 5 pages
        text = page.extract_text()
        match = re.search(r"Item\s+2\.*\s+Management's Discussion.*?(\d{1,3})", text, re.IGNORECASE)
        if match:
            return int(match.group(1))  # Extracted page number
    return None  # If not found

def extract_paragraphs(text, num_paragraphs=2):
    """Extract the first few paragraphs from a given text."""
    paragraphs = re.split(r"\n\s*\n", text)  # Split based on double newlines
    paragraphs = [p.strip() for p in paragraphs if p.strip()]  # Remove empty elements
    return "\n\n".join(paragraphs[:num_paragraphs])  # Return first N paragraphs

def extract_10q_data(pdf_path):
    with pdfplumber.open(pdf_path) as pdf:
        text = "\n".join(page.extract_text() for page in pdf.pages if page.extract_text())
        
        company_name = extract_company_name(text)
        filing_date = extract_filing_date(text)
        
        item_2_page = find_item_2_page(pdf)
        item_2_text = None
        
        if item_2_page:
            item_2_text = extract_paragraphs(pdf.pages[item_2_page - 1].extract_text(), num_paragraphs=2)
        
        return {
            "Company Name": company_name,
            "Filing Date": filing_date,
            "Item 2": item_2_text
        }

if __name__ == "__main__":
    pdf_filename = "Qualcomm-1Q-2024-10-Q.pdf"  # Ensure the file is in the same folder as the script
    extracted_data = extract_10q_data(pdf_filename)
    
    print("Company Name:", extracted_data["Company Name"])
    print("Filing Date:", extracted_data["Filing Date"])
    print("\nItem 2:")
    print(extracted_data["Item 2"])
